import streamlit as st
import pandas as pd
import numpy as np
import plotly.express as px

st.set_page_config(
    page_title="계절과 기온에 따른 소아 호흡기 환자 분석",
    page_icon="🌡️",
    layout="wide"
)

st.title("🌡️ 계절과 기온에 따른 소아 호흡기 환자 분석")
st.markdown(
    "### 계절과 기온 변화를 따라 소아 호흡기 환자 수는 정말 달라질까?"
)
st.write(
    "환자 수와 기온 데이터를 살펴보고, 계절에 따라 소아 호흡기 "
    "환자 수가 어떻게 달라지는지 분석합니다."
)

st.divider()

# -------------------------------
# 1. 데이터 업로드
# -------------------------------
st.header("1. 분석할 데이터 불러오기")

patient_file = st.file_uploader(
    "월별 환자 수 엑셀 파일 (chart.xlsx)",
    type=["xlsx"]
)

weather_file = st.file_uploader(
    "월별 기온 CSV 파일",
    type=["csv"]
)

detail_file = st.file_uploader(
    "질병별·성별·연령별 상세 엑셀 파일 (선택)",
    type=["xlsx"]
)

if patient_file is None or weather_file is None:
    st.info("환자 수 엑셀 파일과 기온 CSV 파일을 업로드해 주세요.")
    st.stop()

# -------------------------------
# 2. 데이터 읽기
# -------------------------------
try:
    patients = pd.read_excel(patient_file)

    patients.columns = patients.columns.astype(str).str.strip()

    if "Category" in patients.columns:
        patients = patients.rename(columns={"Category": "날짜"})

    if "환자수" not in patients.columns or "날짜" not in patients.columns:
        st.error("환자 데이터에 날짜와 환자수 열이 필요합니다.")
        st.stop()

    patients["날짜"] = (
        patients["날짜"].astype(str).str.replace(r"\.0$", "", regex=True)
    )

    patients["환자수"] = pd.to_numeric(
        patients["환자수"], errors="coerce"
    )

    patients = patients.dropna(subset=["환자수"])
    patients["날짜"] = patients["날짜"].str[:6]

    # 기온 CSV 읽기 (기상청 파일 CP949 형식 대응)
    weather_file.seek(0)
    weather = pd.read_csv(
        weather_file,
        encoding="cp949",
        skiprows=11
    )

    weather.columns = weather.columns.astype(str).str.strip()

    weather = weather.rename(columns={
        "일시": "날짜",
        "평균기온(℃)": "평균기온",
        "평균최고기온(℃)": "평균최고기온",
        "평균최저기온(℃)": "평균최저기온"
    })

    weather["날짜"] = (
        weather["날짜"].astype(str).str.replace("-", "", regex=False).str[:6]
    )

    for col in ["평균기온", "평균최고기온", "평균최저기온"]:
        if col in weather.columns:
            weather[col] = pd.to_numeric(weather[col], errors="coerce")

    weather = weather.dropna(subset=["평균기온"])

    # 같은 월이 중복되면 평균으로 정리
    weather = weather.groupby("날짜", as_index=False).agg({
        "평균기온": "mean",
        **({
            "평균최고기온": "mean"
        } if "평균최고기온" in weather.columns else {}),
        **({
            "평균최저기온": "mean"
        } if "평균최저기온" in weather.columns else {})
    })

except Exception as e:
    st.error(f"데이터를 불러오는 중 오류가 발생했습니다: {e}")
    st.stop()

# -------------------------------
# 3. 데이터 기본 정보
# -------------------------------
st.header("2. 데이터 처음 살펴보기")

st.subheader("① 데이터 크기")

col1, col2, col3 = st.columns(3)

col1.metric("환자 데이터 행 개수", f"{len(patients):,}개")
col2.metric("환자 데이터 열 개수", f"{len(patients.columns):,}개")
col3.metric("기온 데이터 행 개수", f"{len(weather):,}개")

st.subheader("② 데이터 앞의 10행")

st.write("월별 환자 수 데이터")
st.dataframe(patients.head(10), use_container_width=True)

st.write("월별 기온 데이터")
st.dataframe(weather.head(10), use_container_width=True)

st.subheader("③ 열마다 자료형 확인")

dtype_df = pd.DataFrame({
    "열 이름": patients.columns,
    "자료형": patients.dtypes.astype(str).values
})

st.dataframe(dtype_df, use_container_width=True)

st.subheader("④ 빈 값 확인")

missing_df = pd.DataFrame({
    "열 이름": patients.columns,
    "빈 값 개수": patients.isnull().sum().values
})

st.dataframe(missing_df, use_container_width=True)

st.write(
    f"환자 데이터 전체 빈 값: **{patients.isnull().sum().sum():,}개**"
)

# -------------------------------
# 4. 환자 수 분포
# -------------------------------
st.header("3. 환자 수 분포 살펴보기")

st.subheader("① 환자 수 기초 통계")

st.dataframe(
    patients["환자수"].describe().to_frame().T,
    use_container_width=True
)

fig = px.histogram(
    patients,
    x="환자수",
    nbins=10,
    title="월별 환자 수 분포",
    labels={"환자수": "환자 수", "count": "월의 개수"}
)

st.plotly_chart(fig, use_container_width=True)

# -------------------------------
# 5. 글자 값 확인
# -------------------------------
st.header("4. 글자 값 확인")

text_cols = patients.select_dtypes(
    include=["object", "string"]
).columns.tolist()

if text_cols:
    selected_text = st.selectbox(
        "확인할 글자형 열 선택",
        text_cols
    )

    st.write("해당 열의 값 종류")
    st.write(patients[selected_text].value_counts(dropna=False))
else:
    st.write("확인할 글자형 열이 없습니다.")

# -------------------------------
# 6. 상세 데이터 분석
# -------------------------------
if detail_file is not None:
    st.header("5. 질병별·성별·연령별 상세 데이터")

    try:
        detail_raw = pd.read_excel(detail_file, header=None)

        st.write("원본 데이터 앞의 10행")
        st.dataframe(detail_raw.head(10), use_container_width=True)

        # 실제 열 이름은 4번째 행
        detail = pd.read_excel(detail_file, header=3)

        detail.columns = detail.columns.astype(str).str.strip()

        st.write("정리된 상세 데이터")
        st.dataframe(detail.head(10), use_container_width=True)

        st.write("상세 데이터 크기")
        st.write(
            f"행: {detail.shape[0]:,}개 / "
            f"열: {detail.shape[1]:,}개"
        )

        st.write("열별 자료형")
        st.dataframe(
            pd.DataFrame({
                "열 이름": detail.columns,
                "자료형": detail.dtypes.astype(str).values
            }),
            use_container_width=True
        )

        st.write("상세 데이터 빈 값")
        st.dataframe(
            detail.isnull().sum().to_frame("빈 값 개수"),
            use_container_width=True
        )

        if "성별구분" in detail.columns:
            st.subheader("성별별 환자 데이터")
            st.write(detail["성별구분"].value_counts())

        if "연령구분5세" in detail.columns:
            st.subheader("연령별 환자 데이터")
            st.write(detail["연령구분5세"].value_counts())

    except Exception as e:
        st.warning(f"상세 데이터 분석 오류: {e}")

# -------------------------------
# 7. 환자 수와 기온 데이터 합치기
# -------------------------------
st.header("6. 환자 수와 기온 데이터 비교")

merged = pd.merge(
    patients,
    weather,
    on="날짜",
    how="inner"
)

merged = merged.dropna(subset=["환자수", "평균기온"])
merged = merged.sort_values("날짜")

if merged.empty:
    st.error("환자 데이터와 기온 데이터의 날짜가 일치하지 않습니다.")
    st.stop()

merged["연도"] = merged["날짜"].str[:4].astype(int)
merged["월"] = merged["날짜"].str[4:6].astype(int)

def get_season(month):
    if month in [3, 4, 5]:
        return "봄"
    elif month in [6, 7, 8]:
        return "여름"
    elif month in [9, 10, 11]:
        return "가을"
    else:
        return "겨울"

merged["계절"] = merged["월"].apply(get_season)

st.metric("비교 가능한 월 개수", f"{len(merged):,}개월")

st.subheader("① 월별 환자 수 변화")

fig = px.line(
    merged,
    x="날짜",
    y="환자수",
    markers=True,
    title="월별 소아 호흡기 환자 수 변화",
    labels={"날짜": "연월", "환자수": "환자 수"}
)

st.plotly_chart(fig, use_container_width=True)

st.subheader("② 월별 평균기온 변화")

fig = px.line(
    merged,
    x="날짜",
    y="평균기온",
    markers=True,
    title="월별 평균기온 변화",
    labels={"날짜": "연월", "평균기온": "평균기온(℃)"}
)

st.plotly_chart(fig, use_container_width=True)

st.subheader("③ 계절별 환자 수 비교")

season_order = ["봄", "여름", "가을", "겨울"]

season_data = (
    merged.groupby("계절", as_index=False)
    .agg(
        평균환자수=("환자수", "mean"),
        평균기온=("평균기온", "mean")
    )
)

season_data["계절"] = pd.Categorical(
    season_data["계절"],
    categories=season_order,
    ordered=True
)

season_data = season_data.sort_values("계절")

fig = px.bar(
    season_data,
    x="계절",
    y="평균환자수",
    color="계절",
    title="계절별 월평균 소아 호흡기 환자 수",
    labels={"평균환자수": "월평균 환자 수", "계절": "계절"}
)

st.plotly_chart(fig, use_container_width=True)

st.dataframe(season_data, use_container_width=True)

st.subheader("④ 기온과 환자 수의 관계")

fig = px.scatter(
    merged,
    x="평균기온",
    y="환자수",
    color="계절",
    hover_data=["날짜"],
    trendline="ols",
    title="평균기온과 소아 호흡기 환자 수의 관계",
    labels={
        "평균기온": "평균기온(℃)",
        "환자수": "환자 수"
    }
)

st.plotly_chart(fig, use_container_width=True)

if len(merged) >= 2:
    correlation = merged["평균기온"].corr(merged["환자수"])

    st.metric(
        "기온과 환자 수의 상관계수",
        f"{correlation:.3f}"
    )

st.divider()

st.header("7. 분석 데이터 확인")
st.dataframe(merged, use_container_width=True)

st.caption(
    "※ 상관관계가 있다고 해서 기온이 환자 수 변화의 직접적인 원인이라고 "
    "단정할 수는 없습니다. 분석 기간과 데이터의 범위도 함께 고려해야 합니다."
)
